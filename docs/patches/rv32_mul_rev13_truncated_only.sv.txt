//=============================================================================
// pangoriscv - rv32 core
// Module : rv32_mul               File : rtl/core/rv32/rv32_mul.v
// Rev    : 1.3                    Date : 2026-10-27
// Brief  : 32x32 multiply, combinational; low product truncated, high product
//          keeps the 33x33 signed array
//=============================================================================

// The delivered core multiplied combinationally with absolute values, four
// cascaded APM blocks and a 64-bit negate: 6.3 ns of the 32.573 ns critical path
// (plan v2 section 1.2).  That cost came from placement, not from the multiply
// itself: it sat between the ALU result and the *next* instruction's ALU operand
// mux, so one instruction's multiply had to finish inside another's execute
// stage.
//
// Here the product is consumed only inside the execute stage and its result is
// registered at the EX/M1 boundary, so nothing downstream depends on it
// combinationally.  One signed 33x33 array covers all four M-extension
// multiplies: extending both operands to 33 bits (sign or zero) makes the
// signed product equal to the unsigned one whenever both are non-negative, and
// the 66-bit result holds every case without a bias correction.
module rv32_mul(
    input  wire [2:0]  op,
    input  wire [31:0] a,
    input  wire [31:0] b,
    output wire [31:0] y
);
    localparam [2:0] MUL_MUL = 3'd0, MUL_H = 3'd1, MUL_HSU = 3'd2, MUL_HU = 3'd3;

    // Two products, deliberately not shared (rev 1.3, plan v27 section 2).
    //
    // The delivered shape built ONE 33x33 signed array and took both the low and
    // the high half out of it, so the *frequent* op (`mul`, which CoreMark issues
    // in crc/state/list code) rode a four-APM serial cascade.  PDS mapped that as
    // u_mul_h/N17_m1..m4 with the partial-product chain m1->m2 and m3->m4 plus a
    // 2.85 ns net between the two sites; measured on the v26 netlist the multiply
    // alone was 1.969 + 1.362 + 2.85 + 1.608 + 1.362 + 0.719 = 9.9 ns, and it sat
    // behind lane 0's ALU (alu_y -> fwd1_h), i.e. 24.3 ns end to end.
    //
    // `mul` only needs bits [31:0], and the low half of a two's-complement
    // product does not depend on how the operands are interpreted, so
    //     prod_signed[31:0] == (a * b)[31:0]     for every a,b
    // Truncating the multiply to its destination width lets the mapper prune the
    // upper partial products: two blocks instead of four, no sign extension, no
    // 64-bit add.  `mulh*` keeps the full 33x33 array and pays for it, but that
    // op is what CoreMark does not issue in the timed loop.
    wire [31:0] lo = a * b;

    // mulhu: both unsigned.  mulhsu: a signed, b unsigned.  mulh: both signed.
    wire signed [32:0] a_x = (op == MUL_HU) ? {1'b0, a} : {a[31], a};
    wire signed [32:0] b_x = (op == MUL_HU || op == MUL_HSU) ? {1'b0, b} : {b[31], b};
    wire signed [65:0] prod = a_x * b_x;

    assign y = (op == MUL_MUL) ? lo : prod[63:32];
endmodule
