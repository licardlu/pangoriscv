//=============================================================================
// pangoriscv - RV32IM processor core + native AXI4 SoC (Pango PG2L100H)
// Module : icache_axi             File : rtl/core/icache_axi.v
// Rev    : 2.0  (v26: BRAM data arrays)   Date : 2026-10-27
// Brief  : 2-way set-associative I-cache (64 sets, 32 B line): hit answered in
//          one cycle, 8-beat AXI4 burst line fill on miss
//=============================================================================

`include "axi_defs.vh"

// ---------------------------------------------------------------------------
// the caches/H1: two-way set-associative instruction cache.
//
// Geometry: 64 sets x 2 ways x 32 B lines = 4096 B.
//   tag       = addr[31:11]   (21 bits)
//   set index = addr[10:5]    (6 bits)
//   word      = addr[4:2]     (3 bits inside the 32 B line)
// Replacement: first invalid way, otherwise the 1-bit LRU victim.
//
// v26 change -- WHY THE DATA ARRAYS MOVED TO BRAM
// ----------------------------------------------
// The delivered 50 MHz implementation closes at 19.563 ns with 82.9% of that
// routed (`evidence/records/optimization/p48e-board50/report_timing.rtr`), and
// every build above 54 MHz ends its route with ~800 nets permanently unrouted
// while sitting at 25% LUT and 9% distributed-RAM utilisation.  A route that
// never converges at a quarter of the device is not a capacity problem: it is
// the 1,742 single-bit distributed-RAM cells (`GTP_RAM64X1DP`) that the two
// caches scatter over the whole die, each needing ten address/enable pins, so
// one write-address bit ends up as a fanout-256 net that costs 2.5 ns alone.
// Moving the arrays into DRM blocks deletes that class of net outright.
//
// The price is that a DRM read is registered: the address must be stable one
// cycle before the data is wanted.  For the I-cache that is free, because the
// array address depends only on the *request* address, never on the tag
// compare.  So the structure becomes
//
//   cycle N    : accept req_addr, latch the decode, drive the bank read index
//   cycle N+1  : bank data + tag compare + way mux -> rsp_valid (combinational)
//
// which presents the response on exactly the same cycle as the old
// accept-edge register did (rsp_valid was a register loaded on the accept
// edge, i.e. visible in cycle N+1).  External timing, the one-lookup-per-cycle
// hit throughput and `req_pair_ahead` (which must stay combinational on
// req_addr because the IFU advances its prefetch pointer with it) are all
// unchanged.  The tag/valid/LRU/successor arrays stay in distributed RAM on
// purpose: they still have to answer `req_pair_ahead` in the request cycle.
//
// A miss keeps the original serialised AR/FILL/MISS_RD path, plus one wait
// state so the bank output reflects the last fill beat before it is sampled.
//
// FENCE.I stays an illegal instruction (plan section 15 option), so the only
// invalidation sources are reset and the explicit `invalidate` pulse, which the
// SoC drives from the halt path: JTAG/down-loader writes followed by a resume
// therefore always see fresh code.
// ---------------------------------------------------------------------------
module icache_axi (
    input wire clk, rst_n,
    input wire invalidate,
    // engine-side lookup interface (identical to axi_read_engine)
    input wire req_valid,
    output wire req_ready,
    input wire [31:0] req_addr,
    input wire [2:0] req_size,
    input wire [7:0] req_len,
    input wire [3:0] req_id,
    output wire rsp_valid,
    input wire rsp_ready,
    output wire [31:0] rsp_data,
    // Second word of the requested 8 B pair.  rsp_pair=0 when the pair would
    // cross the 32 B line (word offset 7), in which case the IFU issues single.
    output wire [31:0] rsp_data_hi,
    output wire rsp_pair,
    // Combinational, valid while the request handshake is in progress: "the
    // response to THIS request will carry two words".  The IFU must know this
    // in the same cycle it raises the request, because it decides there whether
    // the prefetch pointer advances by 4 or by 8.  Both sides read the same
    // value on the same edge, so they cannot disagree.  This is why the tag
    // array stays an asynchronous (distributed-RAM) read.
    output wire req_pair_ahead,
    output wire [1:0] rsp_resp,
    output wire [3:0] rsp_id,
    output reg protocol_fault,
    output reg config_fault,
    // one-cycle performance pulses (used when PERF_ENABLE wires them up)
    output reg miss_pulse,
    output reg access_pulse,
    // AXI4 read master
    output wire m_arvalid,
    input wire m_arready,
    output wire [31:0] m_araddr,
    output wire [3:0] m_arid,
    output wire [7:0] m_arlen,
    output wire [2:0] m_arsize,
    output wire [1:0] m_arburst,
    output wire m_arlock,
    output wire [3:0] m_arcache,
    output wire [2:0] m_arprot,
    output wire [3:0] m_arqos,
    input wire m_rvalid,
    output wire m_rready,
    input wire [31:0] m_rdata,
    input wire [3:0] m_rid,
    input wire [1:0] m_rresp,
    input wire m_rlast
);
    // 4 KB variant: 64 sets x 2 ways x 32 B.  The 8 KB/128-set geometry is what
    // the dual-issue build used first, but the 25k-LUT design could not be
    // routed on this PDS flow; 4 KB recovers ~2.2k LUT / 2.6k FF.  Score trade-
    // off must be measured, not assumed (see evidence: P20 4 KB vs P21 8 KB).
    localparam SETS=64, WORDS=8, LINE_BEATS=8;  // 64 x 2 x 32 B = 4096 B
    localparam IC_IDLE=3'd0, IC_AR=3'd1, IC_FILL=3'd2,
               IC_MISS_W=3'd3, IC_MISS_RD=3'd4, IC_FAULT=3'd5;

    // ---- data array (DRM) ---------------------------------------------------
    // ONE array of SETS*2*WORDS 32-bit words with ONE write port and TWO
    // independent read ports.
    //
    // Why not two banks with one port each (the shape the first v26 attempt
    // used): the mapper's DRAM inference merged the two banks into a single
    // 512x72 simple-dual-port block whose ADDRB took only ONE of the two read
    // addresses and whose byte-write-enables only distinguished the beats, so
    // the pair fetch returned the same word twice -- the bitstream programmed
    // and then produced no UART output at all.  With one array and two read
    // addresses the structure cannot be folded: a DRM has two ports, so the
    // write is replicated and the two addresses stay separate.  Evidence of the
    // bad mapping is in build/v26_ic_bram_50/generate_netlist/*_sim.v
    // (DRM_40_918, .ADDRB(e_ra_q), .DOA(even_q), .DOB(odd_q)).
    reg [31:0] word_mem[0:SETS*2*WORDS-1];
    // ---- metadata (distributed RAM: must answer in the request cycle) -------
    reg [20:0] tag_mem  [0:SETS*2-1];         // {way,set}
    reg        valid_mem[0:SETS*2-1];         // {way,set}
    reg        lru_mem  [0:SETS-1];           // 1 = way1 is the LRU victim

    // ---- cross-line pair side arrays (plan v7 section 1) -------------------
    // Word offset 7 needs (word7 of line L, word0 of line L+32B).  The two words
    // live in two different sets, so the banked data array cannot serve them in
    // one cycle.  These two arrays hold {tag, word0} of whatever line sits in
    // way0/way1 of a set, indexed by that set alone: reading both gives the
    // linear successor of any line without a second port on the main array.
    // `succv0/succv1` mirror valid_mem for exactly those entries and are the
    // only part that has to be reset/invalidated, which a plain FF array can do
    // in one cycle (a distributed RAM cannot).
    reg [52:0] succ0_mem[0:SETS-1];           // {tag[20:0], word0[31:0]}
    reg [52:0] succ1_mem[0:SETS-1];
    reg        succv0[0:SETS-1];
    reg        succv1[0:SETS-1];
    reg [31:0] fill_w0_q;                      // beat 0 of the line being filled

    reg [2:0] state;
    reg [3:0] id_q;
    reg [5:0] set_q;
    reg [2:0] word_q;
    reg [20:0] tag_q;
    reg way_q;
    reg [31:0] rsp_data_q;
    reg [31:0] rsp_data_hi_q;
    reg        rsp_pair_q;
    reg [1:0] rsp_resp_q;
    reg [2:0] beat_q;
    reg fill_bad_q, fill_kill;
    reg [1:0] fill_resp_q;
    integer i;

    // ---- the look pipe ------------------------------------------------------
    reg        look_valid_q;      // the request accepted last cycle is being read
    reg        hit_r;             // ... and it hits
    reg        hit_way_r;         // ... in this way
    reg        pair_r;            // ... and its response carries two words
    reg [31:0] xword_q;           // successor-line word fetched in the accept cycle
    reg        hold_valid;        // a response is held in the rsp_*_q registers
    // ---- bank read pipeline -------------------------------------------------
    reg [31:0] lo_q, hi_q;        // the requested word and its linear successor

    // ========================================================= request decode
    wire [5:0] req_set      = req_addr[10:5];
    wire [20:0] req_tag     = req_addr[31:11];
    wire [2:0] hit_word_off = req_addr[4:2];
    wire hit0 = valid_mem[{1'b0,req_set}] && (tag_mem[{1'b0,req_set}]==req_tag);
    wire hit1 = valid_mem[{1'b1,req_set}] && (tag_mem[{1'b1,req_set}]==req_tag);
    wire       hit_way_now = hit0 ? 1'b0 : 1'b1;
    wire [2:0] req_word_hi = (hit_word_off==3'd7) ? 3'd7 : hit_word_off + 3'd1;
    wire [9:0] hit_lo_index = {req_set, hit_way_now, hit_word_off};
    wire [9:0] hit_hi_index = {req_set, hit_way_now, req_word_hi};

    // Cross-line successor lookup.  nxt_set/nxt_tag describe the 32 B line that
    // starts 8 B after this request, i.e. exactly the line word 8 lives in.
    // The set index wraps inside the cache, so the tag has to take the carry.
    wire [5:0]  nxt_set  = req_set + 6'd1;
    wire [20:0] nxt_tag  = req_tag + {20'd0,(req_set=={6{1'b1}})};
    wire [52:0] succ0_w  = succ0_mem[nxt_set];
    wire [52:0] succ1_w  = succ1_mem[nxt_set];
    wire        succ_ok0 = succv0[nxt_set] && (succ0_w[52:32]==nxt_tag);
    wire        succ_ok1 = succv1[nxt_set] && (succ1_w[52:32]==nxt_tag);
    wire [31:0] succ_word0 = succ_ok1 ? succ1_w[31:0] : succ0_w[31:0];
    // A cross-line pair is only safe when the current line hits (otherwise the
    // miss path answers, and it returns a single word at offset 7) and the
    // successor line is resident with the expected tag.
    wire        xline_ok  = (hit_word_off==3'd7) && (hit0||hit1) && (succ_ok0||succ_ok1);
    wire        pair_now  = (hit_word_off!=3'd7) || xline_ok;
    assign      req_pair_ahead = pair_now;

    // The line address is {tag,set,5'b0}: the 32 B line the requested word is in.
    wire [31:0] line_addr = {tag_q, set_q, 5'b0};

    // ========================================================= bank addresses
    // The miss path wants the *requested word* of the line being filled, so its
    // index is built from the latched request fields; a lookup wants the pair
    // indices of the incoming address.  Only one of the two is ever active.
    wire [2:0]  rd_word_hi = (word_q==3'd7) ? 3'd7 : word_q + 3'd1;
    wire [9:0]  rd_lo_index = {set_q, way_q, word_q};
    wire [9:0]  rd_hi_index = {set_q, way_q, rd_word_hi};
    wire mem_phase = (state==IC_AR) || (state==IC_FILL) ||
                     (state==IC_MISS_W) || (state==IC_MISS_RD);
    wire [9:0] lo_a_d = mem_phase ? rd_lo_index : hit_lo_index;
    wire [9:0] hi_a_d = mem_phase ? rd_hi_index : hit_hi_index;

    // ========================================================= fill / write
    wire fill_beat = (state==IC_FILL) && m_rvalid && (m_rid==id_q);
    wire fill_we   = fill_beat && !fill_kill;
    wire [9:0] fill_idx = {set_q, way_q, beat_q};

    // One write port (only during a fill), two reads latched every cycle: the
    // read ADDRESS is combinational (so an accept at cycle N is in the array at
    // the N/N+1 edge and its data is valid during N+1, which is exactly the
    // latency the old accept-edge register had), and the DATA is registered.
    // Read-during-write returns the old data, which is why IC_MISS_W exists.
    always @(posedge clk) begin
        if (fill_we) word_mem[fill_idx] <= m_rdata;
        lo_q <= word_mem[lo_a_d];
        hi_q <= word_mem[hi_a_d];
    end

    // ========================================================= look resolution
    // During the cycle after an accept the array outputs and the registered tag
    // compare are valid at the same time, so the way mux is all that is left.
    wire        look_hit     = look_valid_q && hit_r;
    wire [31:0] look_word    = lo_q;
    wire [31:0] look_word_hi = (word_q==3'd7) ? xword_q : hi_q;
    wire miss_way = !valid_mem[{1'b0,set_q}] ? 1'b0 :
                    (!valid_mem[{1'b1,set_q}] ? 1'b1 : lru_mem[set_q]);

    // ========================================================= engine responses
    // One response source is active at a time: the look pipe (a hit), or the
    // holding registers (a hit whose consumer was not ready, or a refill).
    assign rsp_valid   = rst_n && (hold_valid || look_hit);
    assign rsp_data    = hold_valid ? rsp_data_q    : look_word;
    assign rsp_data_hi = hold_valid ? rsp_data_hi_q : look_word_hi;
    assign rsp_pair    = hold_valid ? rsp_pair_q    : pair_r;
    assign rsp_resp    = hold_valid ? rsp_resp_q    : `AXI_RESP_OKAY;
    assign rsp_id      = id_q;

    // A new lookup may be taken when the pipe is empty, or when the lookup it
    // currently holds hits AND its response leaves on this edge (that is the
    // one-lookup-per-cycle case).  A miss stalls acceptance: its response comes
    // from the fill, and only one request may be in flight.
    wire pipe_free = !look_valid_q || (look_hit && rsp_ready);
    wire hold_free = !hold_valid   || rsp_ready;
    assign req_ready = rst_n && (state==IC_IDLE) && pipe_free && hold_free;

    assign m_arvalid = rst_n && state==IC_AR;
    assign m_araddr = line_addr;
    assign m_arid = id_q;
    assign m_arlen = 8'd7;                     // 8 beats == one 32 B line
    assign m_arsize = 3'd2;                    // 4 B per beat
    assign m_arburst = `AXI_BURST_INCR;
    assign m_arlock = 1'b0;
    assign m_arcache = 4'b0000;
    assign m_arprot = 3'b000;
    assign m_arqos = 4'b0000;
    assign m_rready = rst_n && state==IC_FILL;

    wire beat_bad = (m_rresp!=`AXI_RESP_OKAY);
    wire line_bad = fill_bad_q || beat_bad;

    always @(posedge clk) begin
        if(!rst_n) begin
            state<=IC_IDLE;
            id_q<=0; set_q<=0; word_q<=0; tag_q<=0; way_q<=0;
            rsp_data_q<=0; rsp_data_hi_q<=0; rsp_pair_q<=0; rsp_resp_q<=0;
            beat_q<=0;
            fill_bad_q<=0; fill_kill<=0; fill_resp_q<=0;
            protocol_fault<=0; config_fault<=0; miss_pulse<=0; access_pulse<=0;
            look_valid_q<=0; hit_r<=0; hit_way_r<=0; pair_r<=0; xword_q<=0;
            hold_valid<=0;
            // Only the VALID bits are reset.  tag_mem is deliberately left
            // uninitialised: a distributed-RAM array cannot be cleared entry by
            // entry, and that per-entry clear was what kept the tag array out of
            // LUTRAM.  Correctness is unchanged: a line is only a hit when
            // valid_mem is set, and valid is only set on the fill edge that also
            // writes its tag.
            for(i=0;i<SETS*2;i=i+1) valid_mem[i]<=0;
            for(i=0;i<SETS;i=i+1) lru_mem[i]<=0;
            for(i=0;i<SETS;i=i+1) begin succv0[i]<=0; succv1[i]<=0; end
            fill_w0_q<=0;
        end else begin
            miss_pulse<=0;
            access_pulse<=0;

            // ------------------------------------------------- accept (cycle N)
            look_valid_q <= (req_valid && req_ready);
            if (req_valid && req_ready) begin
                id_q      <= req_id;
                set_q     <= req_set;
                word_q    <= req_addr[4:2];
                tag_q     <= req_tag;
                hit_r     <= hit0 || hit1;
                hit_way_r <= hit_way_now;
                pair_r    <= pair_now;
                xword_q   <= succ_word0;
            end

            // --------------------------------------- resolve the look (cycle N+1)
            if (look_valid_q) begin
                if (hit_r) begin
                    // the way that answered is now the most recently used one
                    access_pulse<=1'b1;
                    lru_mem[set_q]<= hit_way_r ? 1'b0 : 1'b1;
                    if (!rsp_ready) begin
                        hold_valid    <= 1'b1;
                        rsp_data_q    <= look_word;
                        rsp_data_hi_q <= look_word_hi;
                        rsp_pair_q    <= pair_r;
                        rsp_resp_q    <= `AXI_RESP_OKAY;
                    end
                end else begin
                    miss_pulse<=1'b1;
                    way_q     <= miss_way;
                    beat_q    <= 3'd0;
                    fill_bad_q<=1'b0;
                    fill_resp_q<=2'd0;
                    fill_kill <=1'b0;
                    state     <= IC_AR;
                end
            end

            if (hold_valid && rsp_ready) hold_valid <= 1'b0;

            if(invalidate) begin
                for(i=0;i<SETS*2;i=i+1) valid_mem[i]<=0;
                // The side arrays keep their data (a distributed RAM cannot be
                // cleared entry by entry) and are invalidated through these two
                // FF arrays, which do clear in one cycle.  `invalidate` is a
                // level in this SoC (halt/download/resume), so a level clear is
                // enough: it is still high on every cycle a fill could race it.
                for(i=0;i<SETS;i=i+1) begin succv0[i]<=0; succv1[i]<=0; end
                // A burst in flight must not re-validate the line it is filling.
                if(state==IC_AR || state==IC_FILL) fill_kill<=1;
            end

            case(state)
                IC_IDLE: ;      // acceptance and look resolution are above
                IC_AR: if(m_arvalid && m_arready) state<=IC_FILL;
                IC_FILL: if(m_rvalid) begin
                    if(m_rid!=id_q) begin
                        // Lost framing is a protocol fault, never a data beat.
                        protocol_fault<=1;
                        state<=IC_FAULT;
                    end else begin
                        if(beat_bad) begin
                            fill_bad_q<=1;
                            fill_resp_q<=m_rresp;
                        end
                        // Beat 0 is this line's word 0 -- the very word a
                        // cross-line pair needs as its second half.
                        if(beat_q==3'd0 && !fill_kill) fill_w0_q<=m_rdata;
                        if(m_rlast) begin
                            if(beat_q!=3'd7) begin
                                protocol_fault<=1;
                                state<=IC_FAULT;
                            end else begin
                                if(!fill_kill && !line_bad) begin
                                    tag_mem[{way_q,set_q}]<=tag_q;
                                    valid_mem[{way_q,set_q}]<=1'b1;
                                    // Publish {tag, word0} for this {set,way} so
                                    // the line BEFORE it can answer a
                                    // cross-line pair out of the side array.
                                    if(!way_q) begin
                                        succ0_mem[set_q]<={tag_q,fill_w0_q};
                                        succv0[set_q]<=1'b1;
                                    end else begin
                                        succ1_mem[set_q]<={tag_q,fill_w0_q};
                                        succv1[set_q]<=1'b1;
                                    end
                                end
                                lru_mem[set_q]<=~way_q;
                                // A failed beat is returned to the IFU as a fetch
                                // fault; the line stays invalid.
                                if(line_bad) rsp_resp_q<=(beat_bad ? m_rresp : fill_resp_q);
                                else rsp_resp_q<=`AXI_RESP_OKAY;
                                state<=IC_MISS_W;
                            end
                        end else begin
                            beat_q<=beat_q+3'd1;
                        end
                    end
                end
                // One cycle so the bank output reflects the last fill beat
                // (read-during-write returns the old word otherwise).
                IC_MISS_W: state<=IC_MISS_RD;
                IC_MISS_RD: begin
                    hold_valid    <= 1'b1;
                    rsp_data_q    <= lo_q;
                    rsp_data_hi_q <= (word_q!=3'd7) ? hi_q : 32'd0;
                    rsp_pair_q    <= (word_q!=3'd7);
                    state<=IC_IDLE;
                end
                IC_FAULT: state<=IC_FAULT;
                default: begin
                    protocol_fault<=1;
                    state<=IC_FAULT;
                end
            endcase
        end
    end
endmodule
